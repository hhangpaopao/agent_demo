# deep_pro 依赖清单（云服务器最小集：不含 mineru[all]）
#
# 重要说明：
# - 本项目用 uv 管理，权威依赖源是 pyproject.toml（已列全所有依赖）。
# - `uv add <包>` 是往 pyproject.toml 追加依赖，不会读本文件。
# - 本文件供两种场景参考：① pip / `uv pip install -r requirements.txt`；② 审阅要装什么。
# - 云服务器推荐直接：`uv sync --no-install-package mineru`（跳过 GB 级 torch+模型）。
# - 要全量（含 mineru PDF 解析）：把文末 mineru 那行取消注释，或 `uv sync`。
#
# 分区与 pyproject.toml 一致，版本约束由 uv 解析，此处不锁死版本。

# --------------------------- Web 服务 ---------------------------
fastapi[standard]              # REST + WebSocket
uvicorn[standard]              # ASGI 服务器
python-multipart               # 文件上传 (UploadFile)

# --------------------------- Agent 编排 ---------------------------
deepagents                     # 主智能体 + subagents 框架
langchain                      # LLM 抽象
langchain-openai               # OpenAI 兼容模型适配
openai                         # LLM / VLM / embedding 客户端
langgraph                      # 状态图编排
langgraph-checkpoint-postgres  # PG checkpointer（重启恢复 / 中断续跑）
psycopg[binary,pool]           # PG 异步连接池

# --------------------------- 缓存 ---------------------------
redis                          # 检索结果短时缓存

# --------------------------- 检索 / 数据库 ---------------------------
tavily-python                  # 网络搜索
sqlalchemy                     # ORM（辅助）
asyncmy                        # 异步 MySQL 驱动（问数图）
mysql-connector-python         # 同步 MySQL 驱动（db_tools 直查）
qdrant-client                  # 向量召回（问数图）
elasticsearch[async]>=8,<9      # 关键词召回（问数图）；服务端 8.14，勿升 9.x
jieba>=0.42.1                  # 中文分词：问数图关键词提取（vendor/wenshu 节点依赖）

# --------------------------- RAGFlow / 对象存储 ---------------------------
ragflow-sdk                    # 知识库检索 / dataset 管理
minio                          # 用户上传图片存储

# --------------------------- 文档处理 / 导出 ---------------------------
markdown                       # md → html 中间态
pywin32; sys_platform == "win32"   # 仅 Windows 本地开发；Linux 云服务器自动跳过
# mineru[all]                   # PDF → MD 本地 CLI（含 torch + 模型，GB 级；云服务器装前先确认磁盘/网络）

# --------------------------- 配置 / 日志 / 工具 ---------------------------
omegaconf                      # 分层配置 + ${oc.env:} 环境变量插值
pyyaml                         # YAML 解析
loguru                         # 结构化日志（session_id 注入）
python-dotenv                  # 读取 .env
httpx                          # 异步 HTTP
cryptography                   # 加密依赖

# --------------------------- 测试 ---------------------------
pytest
